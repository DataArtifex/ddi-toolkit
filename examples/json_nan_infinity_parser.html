/**
 * Scientific JSON Parser with IEEE-754 NaN & Infinity Support
 * Zero dependencies • Native JSON.parse reviver architecture
 */

// 1. Parse JSON with support for quoted and bare NaN / Infinity
export function parseScientificJSON(jsonText, options = {}) {
  const {
    parseQuotedNaN = true,
    parseQuotedInfinity = true,
    parseNullAsNaN = false,
    sanitizeBareTokens = true,
    caseInsensitive = true,
  } = options;

  let processed = jsonText;

  // Pre-process unquoted bare tokens (NaN, Infinity, -Infinity) if present
  if (sanitizeBareTokens) {
    processed = processed.replace(
      /("(?:\\.|[^"\\])*")|(\bNaN\b|\b[+-]?Infinity\b|\b[+-]?inf\b)/gi,
      (match, quotedStr, bareToken) => {
        if (quotedStr) return quotedStr;
        if (bareToken) {
          const low = bareToken.toLowerCase();
          if (low.includes('nan')) return '"NaN"';
          if (low.startsWith('-')) return '"-Infinity"';
          return '"Infinity"';
        }
        return match;
      }
    );
  }

  // Native reviver pass
  return JSON.parse(processed, (key, value) => {
    if (parseNullAsNaN && value === null) {
      return NaN;
    }

    if (typeof value === 'string') {
      const testVal = caseInsensitive ? value.toLowerCase().trim() : value.trim();

      // Check NaN patterns
      if (parseQuotedNaN) {
        if (testVal === 'nan' || testVal === '1.#qnan' || testVal === '1.#ind') {
          return NaN;
        }
      }

      // Check Infinity patterns
      if (parseQuotedInfinity) {
        if (testVal === 'infinity' || testVal === '+infinity' || testVal === 'inf' || testVal === '+inf') {
          return Infinity;
        }
        if (testVal === '-infinity' || testVal === '-inf') {
          return -Infinity;
        }
      }
    }

    return value;
  });
}

// 2. Lossless serializer preserving NaN & Infinity as quoted tokens
export function stringifyScientificJSON(obj, replacer = null, space = 2) {
  return JSON.stringify(obj, (key, value) => {
    if (typeof value === 'number') {
      if (Number.isNaN(value)) return 'NaN';
      if (value === Infinity) return 'Infinity';
      if (value === -Infinity) return '-Infinity';
    }
    if (typeof replacer === 'function') {
      return replacer(key, value);
    }
    return value;
  }, space);
}

    


    
    

      

        
📐 RFC 8259 vs IEEE-754

        

          The official JSON standard (RFC 8259) restricts numbers to finite decimal representation: -?(0|[1-9][0-9]*)(\.[0-9]+)?([eE][+-]?[0-9]+)?.
        

        
    
          
  • Bare NaN, Infinity, and -Infinity are invalid JSON grammar.
    • 
          
  • Standard JSON.stringify() coerces NaN & ±∞ to null.
    • 
          
  • Quoting special floats ("NaN") guarantees 100% standards compliance across all HTTP gateways.
    •
